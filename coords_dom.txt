  function measure(id) {
    var root = document.getElementById(id);
    var blocks = root.querySelectorAll('.chord-block');
    var result = [];
    blocks.forEach(function(b) {
      var r = b.getBoundingClientRect();
      result.push(Math.round(r.top) + ':' + Math.round(r.left));
    });
    return result.join(' | ');
  }
