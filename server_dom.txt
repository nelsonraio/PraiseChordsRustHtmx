    document.addEventListener('alpine:init', () => {
      Alpine.data('favButton', (id, initial) => ({
        fav: initial === true || initial === 'true',
        async toggle() {
          const prev = this.fav;
          this.fav = !this.fav; // optimistic
          try {
            const res = await fetch(`/api/songs/${id}/favorite`, { method: 'POST' });
            const json = await res.json();

          } catch (e) {
            this.fav = prev;

      Alpine.data('searchBox', () => ({
        history: [],
        key: 'pc_search_history',
        init() { this.load(); },
        load() {
          try {
            const raw = localStorage.getItem(this.key);
            this.history = raw ? JSON.parse(raw) : [];
          } catch (e) { this.history = []; }
        },
        save() {
          localStorage.setItem(this.key, JSON.stringify(this.history.slice(0,10)));
        },
        add(q) {
          q = (q||'').trim();
          if (!q) return;
          this.history = this.history.filter(x => x !== q);
          this.history.unshift(q);
          this.history = this.history.slice(0,10);
          this.save();
        },
        clear() {
          this.history = [];
          localStorage.removeItem(this.key);
        },
        set(q) {
          this.add(q);
          const el = this.$root.querySelector('#search-input');
          if (el) {
            el.value = q;
            this.$nextTick(() => {
              el.dispatchEvent(new Event('input', { bubbles: true }));
              el.dispatchEvent(new KeyboardEvent('keyup'));
            });
          }
        }
      }));

    window.praiseChordsYoutube = {
      currentVideoId: '',
      isMobile() {
        return /Android|iPhone|iPad|iPod|Mobile|Opera Mini|IEMobile/i.test(navigator.userAgent || '');
      },
      extractVideoId(url) {
        let value = String(url || '').trim();
        if (!value) return '';
        if (/^[A-Za-z0-9_-]{11}$/.test(value)) return value;
        if (!/^https?:\/\//i.test(value)) value = 'https://' + value;
        try {
          const parsed = new URL(value);
          let id = parsed.searchParams.get('v');
          if (!id && parsed.hostname === 'youtu.be') id = parsed.pathname.slice(1);
          if (!id && parsed.pathname.startsWith('/embed/')) id = parsed.pathname.split('/')[2];
          if (!id && parsed.pathname.startsWith('/shorts/')) id = parsed.pathname.split('/')[2];
          if (!id && parsed.pathname.startsWith('/live/')) id = parsed.pathname.split('/')[2];
          return /^[A-Za-z0-9_-]{11}$/.test(id || '') ? id : '';
        } catch (e) {
          return '';
        }
      },
      play(url) {
        const videoId = this.extractVideoId(url);
        if (!videoId) return;
        this.currentVideoId = videoId;
        const watchUrl = `https://www.youtube.com/watch?v=${videoId}`;
        // No mobile, o YouTube bloqueia a reprodu├º├úo em iframe para v├¡deos
        // com embedding desativado. Abre diretamente no app/site do YouTube.
        if (this.isMobile()) {
          window.open(watchUrl, '_blank', 'noopener,noreferrer');
          return;
        }
        document.getElementById('youtube-frame').src = `https://www.youtube.com/embed/${videoId}`;
        document.getElementById('youtube-player').classList.remove('hidden');
      },
      close() {
        this.currentVideoId = '';
        document.getElementById('youtube-frame').src = '';
        document.getElementById('youtube-player').classList.add('hidden');
      }
    };

    function openModal() {
      const modal = document.getElementById('modal');
      modal.classList.remove('hidden');
      // When displaying the chord viewer, remove the dark overlay, padding,
      // and container constraints so chords use the maximum space
      // with no black border
      if (modal.querySelector('[x-data*="chordViewer"]')) {
        modal.classList.remove('bg-black', 'bg-opacity-70', 'p-2', 'p-4');
        modal.style.padding = '0';
        modal.style.overflowY = 'hidden';
        modal.style.alignItems = 'stretch';
        modal.style.justifyContent = 'flex-start';
      } else {
        modal.classList.add('bg-black', 'bg-opacity-70', 'p-2');
        modal.style.padding = '';
        modal.style.overflowY = 'auto';
        modal.style.alignItems = '';
        modal.style.justifyContent = '';
      }
      document.body.style.overflow = 'hidden';
    }

    function closeModal() {
      const modal = document.getElementById('modal');
      modal.classList.add('hidden');
      document.body.style.overflow = '';
      // Reset modal styling for non-chord-viewer content
      modal.classList.add('bg-black', 'bg-opacity-70', 'p-2');
      modal.style.padding = '';
      modal.style.overflowY = 'auto';
      modal.style.alignItems = '';
      modal.style.justifyContent = '';
    }

    document.body.addEventListener('htmx:afterSwap', (event) => {
